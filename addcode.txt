import requests
from flask import Flask, jsonify, request  # <-- Make sure 'request' is here
app = Flask(__name__)

# Sample data: 5 students
students = [
    {"id": 1, "name": "Juan Dela Cruz", "course": "BSIT", "year": 4},
    {"id": 2, "name": "Maria Santos", "course": "BSIT", "year": 3},
    {"id": 3, "name": "Pedro Penduko", "course": "BSCS", "year": 2},
    {"id": 4, "name": "Ana Reyes", "course": "BSIS", "year": 1},
    {"id": 5, "name": "Luis Manzano", "course": "BSIT", "year": 4}
]

# Endpoint 1: Retrieve all students
@app.route('/students', methods=['GET'])
def get_all_students():
    return jsonify(students)

# Endpoint 2: Retrieve a specific student by ID
@app.route('/students/<int:student_id>', methods=['GET'])
def get_student(student_id):
    # Search for the student in the list
    student = next((s for s in students if s["id"] == student_id), None)
    
    if student:
        return jsonify(student), 200
    else:
        # Error Handling: Student not found
        return jsonify({"error": "Student not found"}), 404

import requests

BASE_URL = "http://127.0.0.1:5000/students"

def main():
    print("===== STUDENT INFORMATION CLIENT =====")
    
    while True:
        try:
            #modify here
            print("\n\nType Student ID or type 'add' to add a new student (or 'q' to quit):")
            UserInput = input("\nInput: ")
            
            if UserInput.lower() == 'q':
                break

            elif UserInput.lower() == 'add':
                print("Enter student details")
                name = input("name: ")
                course = input("course: ")
                year = input("year level: ")
                
                AddStudent = {
                "name": name, 
                "course": course, 
                "year": int(year)
                }
                print("Sending requeest")

                response = requests.post(BASE_URL, json=AddStudent)

                if response.status_code == 201:
                    print("Add success")
                    print(response.json())

                else:
                    print("Add failed")
                    print(response.json())
                
                continue 
                
                print("Connecting to Student Web Service...")
            
            # Send HTTP GET Request to Endpoint 2
            response = requests.get(f"{BASE_URL}/{UserInput}")
            
            # Check the HTTP status code
            if response.status_code == 200:
                # Convert JSON response to a Python dictionary
                student = response.json()
                
                print("Student Found!")
                print(f"ID: {student['id']}")
                print(f"Name: {student['name']}")
                print(f"Course: {student['course']}")
                print(f"Year Level: {student['year']}")
                
            elif response.status_code == 404:
                print("Student not found.")
                
            else:
                print(f"Unexpected error. HTTP Status: {response.status_code}")
                
        except requests.exceptions.ConnectionError:
            print("Error: Could not connect to the server. Is the Flask app running?")
            break
        except ValueError:
            print("Please enter a valid numeric ID.")

if __name__ == '__main__':
    main()
